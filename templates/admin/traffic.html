# Xray-core 推荐：每 5 分钟读取并 reset 统计，按增量累加到会员
python3 scripts/xray_usage_collector.py \
  --backend xray \
  --xray-bin /usr/local/bin/xray \
  --xray-api 127.0.0.1:10085 \
  --reset \
  --report-url {{ base or "https://sub.0909106.xyz" }}/api/traffic/report \
  --token 你的UPDATE_TOKEN
